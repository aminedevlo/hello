async function fetchWithFallback(url) { const encodedUrl = encodeURIComponent(url); // Array of public CORS proxy strategies const proxies = [ { name: "allorigins-json", url: `https://api.allorigins.win/get?url=${encodedUrl}`, parse: async (res) => { const data = await res.json(); return data.contents; } }, { name: "allorigins-raw", url: `https://api.allorigins.win/raw?url=${encodedUrl}`, parse: (res) => res.text() }, { name: "corsproxy-io", url: `https://corsproxy.io/?${encodedUrl}`, parse: (res) => res.text() }, { name: "codetabs", url: `https://api.codetabs.com/v1/proxy?quest=${encodedUrl}`, parse: (res) => res.text() } ]; let lastError = null; for (const proxy of proxies) { try { console.log(`Attempting request via ${proxy.name}...`); const controller = new AbortController(); const timeoutId = setTimeout(() => controller.abort(), 6000); // 6s timeout const response = await fetch(proxy.url, { signal: controller.signal }); clearTimeout(timeoutId); if (!response.ok) { throw new Error(`HTTP ${response.status} from ${proxy.name}`); } const html = await proxy.parse(response); if (html && html.length > 50) { return html; // Successfully retrieved markup } throw new Error(`Empty response from ${proxy.name}`); } catch (err) { console.warn(`Proxy ${proxy.name} failed:`, err.message); lastError = err; } } throw new Error(`All proxies failed. Root issue: ${lastError ? lastError.message : "Network unreachable"}`); }
